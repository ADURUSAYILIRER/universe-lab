"""UNIVERSE LAB - physics engine v2 (RK4) + Experiment 1 tools. Units: G = 1. Arrays: pos, vel have shape (3 bodies, 3 dims).""" import numpy as np from scipy.integrate import solve_ivp G = 1.0 def acceleration(pos, m): """Newtonian acceleration of each body from the other two.""" acc = np.zeros_like(pos) for i in range(len(m)): for j in range(len(m)): if i != j: r = pos[j] - pos[i] acc[i] += G * m[j] * r / np.linalg.norm(r) ** 3 return acc def rk4_step(pos, vel, m, dt): """One 4th-order Runge-Kutta step for (pos, vel).""" k1v = acceleration(pos, m); k1x = vel k2v = acceleration(pos + 0.5*dt*k1x, m); k2x = vel + 0.5*dt*k1v k3v = acceleration(pos + 0.5*dt*k2x, m); k3x = vel + 0.5*dt*k2v k4v = acceleration(pos + dt*k3x, m); k4x = vel + dt*k3v pos_new = pos + dt/6 * (k1x + 2*k2x + 2*k3x + k4x) vel_new = vel + dt/6 * (k1v + 2*k2v + 2*k3v + k4v) return pos_new, vel_new def energy(pos, vel, m): """Total energy = kinetic + potential.""" kin = 0.5 * np.sum(m[:, None] * vel**2) pot = 0.0 for i in range(len(m)): for j in range(i + 1, len(m)): pot -= G * m[i] * m[j] / np.linalg.norm(pos[i] - pos[j]) return kin + pot def run(pos, vel, m, dt, t_end): """Integrate until t_end. Returns final pos, vel.""" for _ in range(int(round(t_end / dt))): pos, vel = rk4_step(pos, vel, m, dt) return pos, vel def prediction_horizon(pos, vel, m, sigma, dt=1e-3, t_max=20.0, threshold=1.0, seed=0): """Experiment 1 core: observer knows the initial state only up to error sigma. Returns the first time the predicted and true trajectories differ (max body-position error) by more than `threshold`. None if never.""" rng = np.random.default_rng(seed) pos2 = pos + sigma * rng.standard_normal(pos.shape) # observer's wrong state vel2 = vel.copy() t = 0.0 while t < t_max: pos, vel = rk4_step(pos, vel, m, dt) pos2, vel2 = rk4_step(pos2, vel2, m, dt) t += dt if np.max(np.linalg.norm(pos - pos2, axis=1)) > threshold: return t return None def _rhs(t, y, m): pos = y[:9].reshape(3, 3); vel = y[9:].reshape(3, 3) return np.concatenate([vel.ravel(), acceleration(pos, m).ravel()]) def solve_adaptive(pos, vel, m, t_eval, rtol=1e-11, atol=1e-12): """Adaptive DOP853 integration. Returns array (18, len(t_eval)): 9 positions then 9 velocities. Needed for chaotic ICs with close encounters, where fixed-step RK4 fails.""" y0 = np.concatenate([pos.ravel(), vel.ravel()]) sol = solve_ivp(_rhs, (0, t_eval[-1]), y0, method="DOP853", t_eval=t_eval, rtol=rtol, atol=atol, args=(m,)) return sol.y def horizon_adaptive(pos, vel, m, sigma, seed=0, t_max=30.0, dt_out=0.05, threshold=1.0, rtol=1e-11): """Experiment 1 core (adaptive version): observer's initial positions are wrong by sigma * N(0,1). Returns first time the max body-position error exceeds `threshold`, or None if it never does within t_max (censored).""" te = np.arange(0, t_max + dt_out / 2, dt_out) ref = solve_adaptive(pos, vel, m, te, rtol=rtol) rng = np.random.default_rng(seed) alt = solve_adaptive(pos + sigma * rng.standard_normal(pos.shape), vel, m, te, rtol=rtol) err = np.max(np.linalg.norm((ref[:9] - alt[:9]).reshape(3, 3, -1), axis=1), axis=0) if err.max() <= threshold: return None return te[np.argmax(err > threshold)] # ---- initial conditions ---- M = np.ones(3) # Figure-eight orbit (stable, periodic): CONTROL system, lambda ~ 0 _p = np.array([0.97000436, -0.24308753, 0.0]) _v = np.array([-0.93240737, -0.86473146, 0.0]) FIG8_POS = np.array([-_p, _p, [0, 0, 0]]) FIG8_VEL = np.array([-_v/2, -_v/2, _v]) FIG8_PERIOD = 6.32591398 # CHAOTIC system (v2). Chosen by random search (seed 73), verified with DOP853: # E = -1.991, min body distance 0.061 over t<=40, energy drift 1.4e-9, # Lyapunov exponent lambda ~ 0.74 (Benettin, 3 perturbation directions: 0.75/0.77/0.72). # CAUTION: with lambda ~ 0.74 the *numerical* reference itself loses accuracy for t > ~20-25 # (rtol=1e-11 vs 1e-13 differ by 1e-2 at t=20, 5e-2 at t=25). Use t_max <= 30 and sigma >= 1e-7.
